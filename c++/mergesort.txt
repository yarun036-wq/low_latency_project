#include <iostream>
#include<bits/stdc++.h>
using namespace std;
void merger(vector<int>&arr,int low,int mid,int high)
{
   int n1=mid-low+1;
   int n2=high-mid;
   int L[n1+1],M[n2+1];
   L[n1]=INT_MAX,M[n2]=INT_MAX;
   for(int i=0;i<n1;i++)
   {
       L[i]=arr[low+i];
   }
   for(int j=0;j<n2;j++)
   {
       M[j]=arr[j+mid+1];
   }
  int i=0;
  int j=0;
  int k=low;
  while(i<n1 && j<n2)
  {
      if(L[i]<M[j])
      {
          arr[k]=L[i];
          i++;
          k++;
      }
      else
      {
          arr[k]=M[j];
          j++;
          k++;
      }
  }
  while(i<n1)
  {
      arr[k]=L[i];
      i++;
      k++;
  }
  while(j<n2)
  {
      arr[k]=M[j];
      j++;
      k++;
  }
}
void merge(vector<int>&arr,int low,int high)
{
    if(low<high)
    {
        int mid=(low+high)/2;
        merge(arr,low,mid);
        merge(arr,mid+1,high);
        merger(arr,low,mid,high);
    }
}
int main() {
     vector<int>arr={6, 5, 12, 10, 9, 1};
     int n=arr.size();
     merge(arr,0,n-1);
     for(auto x:arr)
     {
         cout<<x<<" ";
     }
     return 0;
}
