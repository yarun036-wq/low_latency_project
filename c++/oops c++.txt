#include<iostream>

using namespace std;
class MyClass
{
    public:
    int myNum;
    string myString;
    MyClass() //everytime constructor whenever new object is created.
    {
        cout<<"Hello"<<endl;
    }
    MyClass(int x,string y) //parameterized constructor
    {
        myNum=x;
        myString=y;
    }
    void myMethod() //Function inside class
    {
        cout<<"Hello World"<<endl;
    }
    int speed(int maxSpeed); // to use function outside we should declare function inside.
};
int MyClass::speed(int maxSpeed) //Function outside class
{
    return maxSpeed;
}
