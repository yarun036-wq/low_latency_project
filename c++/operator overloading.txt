// Online C++ compiler to run C++ program online
#include <iostream>
using namespace std;
class Complex
{
    private:
    int real,imag;
    public:
    Complex()
    {
        
    }
    Complex(int x,int y)
    {
        real=x;
        imag=y;
    }
    Complex operator +(Complex c)
    {
        Complex temp;
        temp.real=real+c.real;
        temp.imag=imag+c.imag;
        return temp;
    }
    void print()
    {
        cout<<real<<"+i"<<imag<<endl;
    }
};
int main() {
   Complex c1(4,5),c2(5,9);
   Complex c3=c1+c2;
   c3.print();
    return 0;
}
